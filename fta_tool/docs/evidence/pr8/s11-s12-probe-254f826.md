## スクリプト（実行したもの）

```python
"""Probe of S-11 / S-12 on a temporary DB (no change to the app).
Run with the current directory set to an empty temporary folder."""
import sqlite3, sys, time
from fastapi.testclient import TestClient
from app.main import app

mode = sys.argv[1]
with TestClient(app) as c:
    def analysis(t):
        r = c.post("/analyses", data={"title": t, "top_event": "事象"}, follow_redirects=False)
        return int(r.headers["location"].rstrip("/").rsplit("/", 1)[1])
    def l1(a, t):
        return c.post(f"/analyses/{a}/nodes/add-level1", json={"title": t}).json()["node_id"]
    def child(p, t):
        return c.post(f"/nodes/{p}/children", json={"title": t}).json()["node_id"]
    db = sqlite3.connect("fta_tool.db")
    def count(a):
        return db.execute("select count(*) from nodes where analysis_id=?", (a,)).fetchone()[0]
    if mode == "s11-analysis":
        x = analysis("X"); y = analysis("Y")
        xa = l1(x, "X一次"); ya = l1(y, "Y一次"); yb = child(ya, "Y二次")
        yc = l1(y, "Y一次（無関係）")
        db.execute("update nodes set parent_id=? where id=?", (xa, yb)); db.commit()  # Y二次 -> X一次 (cross-analysis)
        print("before X", count(x), "Y", count(y))
        r = c.post(f"/analyses/{x}/delete")
        print("delete X:", r.status_code, r.text[:80])
        print("after  X", count(x), "Y", count(y), "Y rows", db.execute("select id,title from nodes where analysis_id=?", (y,)).fetchall())
    elif mode == "s11-node":
        x = analysis("X"); y = analysis("Y")
        xa = l1(x, "X一次"); ya = l1(y, "Y一次"); yb = child(ya, "Y二次")
        db.execute("update nodes set parent_id=? where id=?", (xa, yb)); db.commit()
        r = c.post(f"/nodes/{xa}/delete")
        print("delete X一次:", r.status_code, "Y count", count(y))
    elif mode == "s11-cycle":
        x = analysis("X"); a = l1(x, "A"); b = child(a, "B")
        db.execute("update nodes set parent_id=? where id=?", (b, a)); db.commit()  # A <-> B
        c2 = TestClient(app, raise_server_exceptions=False)
        r = c2.post(f"/analyses/{x}/delete")
        print("delete analysis with cycle:", r.status_code, "nodes", count(x),
              "analysis", db.execute("select count(*) from analyses where id=?", (x,)).fetchone()[0])
    elif mode in ("s12-parent", "s12-noparent"):
        x = analysis("X"); a = l1(x, "A"); b = child(a, "B")
        db.execute("update nodes set parent_id=? where id=?", (b, a))  # A <-> B (cycle)
        db.execute("update nodes set user_judgement='yes'"); db.commit()
        body = {"parent_id": b} if mode == "s12-parent" else {}
        print("sending generate level 3", body, flush=True)
        t = time.time()
        r = c.post(f"/analyses/{x}/generate/level/3", json=body)
        print("answered", r.status_code, round(time.time() - t, 2), "s", r.text[:120])
```

実行の方法（各モードを新しい一時ディレクトリで。`$S` は作業用のフォルダ、`$WT` は対象のクリーンな worktree の `fta_tool`）：

```bash
for m in s11-analysis s11-node s11-cycle s12-parent s12-noparent; do
  d=$(mktemp -d -p "$S/probe")
  (cd "$d" && ulimit -v 3000000 && AI_PROVIDER=mock timeout 20 "$S/venv/bin/python" -I -c \
     "import sys; sys.path.insert(0, '$WT'); sys.argv=['p','$m']; exec(open('$S/probe/probe_s11_s12.py').read())"; echo "exit $?")
done
```
